cmake_minimum_required(VERSION 3.22)
project(GBoy VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GBOY_BUILD_PLUGIN "Build JUCE instrument" ON)
enable_testing()
add_executable(gboy_generate_content tools/GenerateContent.cpp)
target_include_directories(gboy_generate_content PRIVATE src)
add_executable(gboy_core_tests tests/CoreTests.cpp)
target_include_directories(gboy_core_tests PRIVATE src)
add_test(NAME audio_core COMMAND gboy_core_tests)
add_executable(gboy_hybrid_tests tests/HybridTests.cpp)
target_include_directories(gboy_hybrid_tests PRIVATE src)
add_test(NAME hybrid_core COMMAND gboy_hybrid_tests)
add_executable(gboy_realtime_tests tests/RealtimeTests.cpp)
target_include_directories(gboy_realtime_tests PRIVATE src)
add_test(NAME realtime_allocations COMMAND gboy_realtime_tests)
add_executable(gboy_vocal_tests tests/VocalTests.cpp)
target_include_directories(gboy_vocal_tests PRIVATE src)
add_test(NAME vocal_core COMMAND gboy_vocal_tests)
add_executable(gboy_effects_tests tests/EffectsTests.cpp)
target_include_directories(gboy_effects_tests PRIVATE src)
add_test(NAME effects_core COMMAND gboy_effects_tests)
add_executable(gboy_sequencer_tests tests/SequencerTests.cpp)
target_include_directories(gboy_sequencer_tests PRIVATE src)
add_test(NAME sequencer_core COMMAND gboy_sequencer_tests)
if(GBOY_BUILD_PLUGIN)
 set(JUCE_SOURCE_DIR "" CACHE PATH "Path to JUCE 8.0.6 checkout")
 if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" juce)
 else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
 endif()
 set(formats VST3 Standalone)
 if(APPLE)
  list(APPEND formats AU)
 endif()
 juce_add_plugin(GBoy COMPANY_NAME "G Boy Audio" BUNDLE_ID "audio.gboy.instrument" PLUGIN_MANUFACTURER_CODE Gboy PLUGIN_CODE Gb01 FORMATS ${formats} PRODUCT_NAME "G Boy" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(GBoy PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp)
 target_compile_definitions(GBoy PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(GBoy PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
 juce_add_console_app(gboy_plugin_tests PRODUCT_NAME "G Boy processor tests")
 target_sources(gboy_plugin_tests PRIVATE tests/PluginTests.cpp)
 target_include_directories(gboy_plugin_tests PRIVATE src)
 target_link_libraries(gboy_plugin_tests PRIVATE GBoy juce::juce_audio_utils)
 add_test(NAME plugin_processor COMMAND gboy_plugin_tests)
endif()
